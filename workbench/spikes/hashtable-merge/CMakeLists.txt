include("${PROJECT_SOURCE_DIR}/workbench/benchmarks/hashtable/abseil.cmake")
add_executable(hashtable_merge_check EXCLUDE_FROM_ALL check.cpp)
sixdb_target(hashtable_merge_check)
target_link_libraries(hashtable_merge_check PRIVATE ikea::hashtable)
add_test(NAME hashtable_merge_check COMMAND hashtable_merge_check)
set(SIXDB_MERGE_ONLY_MODE 16 CACHE STRING "16 candidate (18 is a historical result label)")
set(SIXDB_MERGE_REDUCER_KIND 0 CACHE STRING "0 byte fold, 1 copy, 2 word add, 3 word fold")
option(SIXDB_MERGE_SIZE_SWEEP "Compile merge footprint crossover fixtures" OFF)
option(SIXDB_MERGE_WIDE_FAMILIES "Compile 96-byte and full-width key fixtures" OFF)
foreach(v IN ITEMS 0 8 64)
  add_executable(hashtable_merge_bench_${v} EXCLUDE_FROM_ALL bench.cpp)
  sixdb_target(hashtable_merge_bench_${v})
  target_compile_definitions(hashtable_merge_bench_${v} PRIVATE NDEBUG MERGE_VALUE_BYTES=${v} MERGE_REDUCER_KIND=${SIXDB_MERGE_REDUCER_KIND})
  if(NOT SIXDB_MERGE_ONLY_MODE STREQUAL "")
    target_compile_definitions(hashtable_merge_bench_${v} PRIVATE MERGE_ONLY_MODE=${SIXDB_MERGE_ONLY_MODE})
  endif()
  target_compile_definitions(hashtable_merge_bench_${v} PRIVATE MERGE_SIZE_SWEEP=$<BOOL:${SIXDB_MERGE_SIZE_SWEEP}> MERGE_WIDE_FAMILIES=$<BOOL:${SIXDB_MERGE_WIDE_FAMILIES}>)
  # Give before/after and Abseil bodies the same alignment. Identical x86
  # instruction streams differed by 5% in the default placement screen.
  target_compile_options(hashtable_merge_bench_${v} PRIVATE -mllvm -inline-threshold=1000 -falign-functions=64)
  target_link_libraries(hashtable_merge_bench_${v} PRIVATE ikea::hashtable hashtable_abseil_baseline)
endforeach()

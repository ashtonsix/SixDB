cmake_minimum_required(VERSION 3.24)
get_filename_component(SIXDB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
include("${SIXDB_ROOT}/cmake/ClangVersion.cmake")
project(OrbitalSystemsProbe LANGUAGES CXX)
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR
   NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL SIXDB_CLANG_VERSION)
  message(FATAL_ERROR "Use pinned Clang ${SIXDB_CLANG_VERSION}")
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -g")
include("${SIXDB_ROOT}/cmake/Targets.cmake")
find_package(Threads REQUIRED)
add_library(view_fixture STATIC ../prototype.cpp ../recycling.cpp)
sixdb_target(view_fixture)
target_include_directories(view_fixture PUBLIC ..)
target_link_libraries(view_fixture PUBLIC Threads::Threads)
add_executable(resource_probe resources.cpp)
sixdb_target(resource_probe)
target_link_libraries(resource_probe PRIVATE view_fixture)
add_executable(journal_probe journal.cpp)
sixdb_target(journal_probe)

add_library(publication_mechanism STATIC ../publication/material.cpp ../publication/journal.cpp)
sixdb_target(publication_mechanism)
target_include_directories(publication_mechanism PUBLIC ../publication)
add_subdirectory("${SIXDB_ROOT}/ikea" ikea)
add_executable(composition_probe composition.cpp)
sixdb_target(composition_probe)
target_link_libraries(composition_probe PRIVATE publication_mechanism ikea::tuplepack Threads::Threads)
add_executable(unit_boundary unit_boundary.cpp)
sixdb_target(unit_boundary)
target_link_libraries(unit_boundary PRIVATE publication_mechanism ikea::bec)
add_executable(intent_probe intent_probe.cpp)
sixdb_target(intent_probe)
target_link_libraries(intent_probe PRIVATE ikea::tuplepack Threads::Threads)
add_executable(binding_probe binding_probe.cpp)
sixdb_target(binding_probe)
target_link_libraries(binding_probe PRIVATE Threads::Threads)

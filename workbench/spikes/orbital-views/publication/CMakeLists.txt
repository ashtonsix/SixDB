cmake_minimum_required(VERSION 3.24)
include(../../../../cmake/ClangVersion.cmake)
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -g" CACHE STRING "Release flags")
project(OrbitalPublication LANGUAGES CXX)
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR
   NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL SIXDB_CLANG_VERSION)
  message(FATAL_ERROR "Use pinned Clang ${SIXDB_CLANG_VERSION}")
endif()
include(../../../../cmake/Targets.cmake)
enable_testing()
add_library(publication_core STATIC material.cpp journal.cpp)
sixdb_target(publication_core)
add_executable(publication_probe probe.cpp)
sixdb_target(publication_probe)
target_link_libraries(publication_probe PRIVATE publication_core)
add_test(NAME publication_probe COMMAND publication_probe)
add_executable(publication_bench bench.cpp)
sixdb_target(publication_bench)
target_link_libraries(publication_bench PRIVATE publication_core)

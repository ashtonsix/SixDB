cmake_minimum_required(VERSION 3.24)
project(orbital_views_residency LANGUAGES CXX)
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "21.1.8")
  message(FATAL_ERROR "Use SixDB's pinned Clang 21.1.8")
endif()
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -g -DNDEBUG")
include("${CMAKE_CURRENT_LIST_DIR}/../../../../cmake/Targets.cmake")
add_library(residency_core STATIC core.cpp memory.cpp)
sixdb_target(residency_core)
add_executable(residency_probe bench.cpp)
sixdb_target(residency_probe)
target_link_libraries(residency_probe PRIVATE residency_core)

cmake_minimum_required(VERSION 3.24)
include(../../../../cmake/ClangVersion.cmake)
if(NOT DEFINED CMAKE_CXX_COMPILER)
  find_program(CONTROL_CLANGXX NAMES clang++-${SIXDB_CLANG_MAJOR} REQUIRED)
  set(CMAKE_CXX_COMPILER "${CONTROL_CLANGXX}" CACHE FILEPATH "Pinned compiler")
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -g" CACHE STRING "Release flags")
project(OrbitalViewControl LANGUAGES CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR
   NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL SIXDB_CLANG_VERSION)
  message(FATAL_ERROR "Requires Clang ${SIXDB_CLANG_VERSION}")
endif()
include(../../../../cmake/Targets.cmake)
find_package(Threads REQUIRED)
enable_testing()
add_library(control_mechanisms STATIC mechanisms.cpp)
sixdb_target(control_mechanisms)
target_link_libraries(control_mechanisms PUBLIC Threads::Threads)
foreach(name probe bench)
  add_executable(control_${name} ${name}.cpp)
  sixdb_target(control_${name})
  target_link_libraries(control_${name} PRIVATE control_mechanisms)
endforeach()
add_test(NAME control_probe COMMAND control_probe)
set_tests_properties(control_probe PROPERTIES TIMEOUT 30)
add_library(cut_mechanisms STATIC cut.cpp)
sixdb_target(cut_mechanisms)
target_link_libraries(cut_mechanisms PUBLIC control_mechanisms)
foreach(name probe bench)
  add_executable(cut_${name} cut_${name}.cpp)
  sixdb_target(cut_${name})
  target_link_libraries(cut_${name} PRIVATE cut_mechanisms)
endforeach()
add_test(NAME cut_probe COMMAND cut_probe)
set_tests_properties(cut_probe PROPERTIES TIMEOUT 30)

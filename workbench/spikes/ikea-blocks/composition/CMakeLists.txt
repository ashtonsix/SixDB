cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(ikea_composition LANGUAGES CXX)
  if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "21.1.8")
    message(FATAL_ERROR "This probe requires Linux Clang 21.1.8")
  endif()
  function(sixdb_target target)
    target_compile_features(${target} PUBLIC cxx_std_23)
    target_compile_options(${target} PRIVATE -O3 -g -fno-fast-math -ffp-contract=off -fdenormal-fp-math=ieee)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF INTERPROCEDURAL_OPTIMIZATION OFF UNITY_BUILD OFF)
  endfunction()
endif()
add_library(ikea_composition_core STATIC graph.cpp lower.cpp prepare.cpp driver.cpp native_exports.cpp
  stages_load.cpp stages_features.cpp stages_model.cpp stages_done.cpp)
sixdb_target(ikea_composition_core)
target_include_directories(ikea_composition_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
  target_compile_options(ikea_composition_core PUBLIC -mavx2)
endif()
add_executable(ikea_composition_check check.cpp)
sixdb_target(ikea_composition_check)
target_link_libraries(ikea_composition_check PRIVATE ikea_composition_core)
add_executable(ikea_composition_bench bench.cpp)
sixdb_target(ikea_composition_bench)
target_link_libraries(ikea_composition_bench PRIVATE ikea_composition_core)

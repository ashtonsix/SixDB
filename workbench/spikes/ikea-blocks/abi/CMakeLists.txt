cmake_minimum_required(VERSION 3.24)
project(ikea_abi_probe LANGUAGES CXX)
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "21.1.8")
  message(FATAL_ERROR "This recorded probe requires Linux Clang 21.1.8")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
set(CMAKE_UNITY_BUILD OFF)
add_library(probe_objects OBJECT producer.cpp caller.cpp sink.cpp)
add_executable(abi_check check.cpp $<TARGET_OBJECTS:probe_objects>)
foreach(target probe_objects abi_check)
  target_compile_options(${target} PRIVATE -O3 -g -fno-fast-math -ffp-contract=off
    -fdenormal-fp-math=ieee -Wno-return-type-c-linkage)
endforeach()

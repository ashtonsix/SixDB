if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    cmake_minimum_required(VERSION 3.24)
    get_filename_component(change_root "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
    include("${change_root}/cmake/ClangVersion.cmake")
    find_program(change_clang NAMES clang++-${SIXDB_CLANG_MAJOR} REQUIRED)
    set(CMAKE_CXX_COMPILER "${change_clang}" CACHE FILEPATH "Pinned compiler")
    project(FrameChangeWitness LANGUAGES CXX)
    set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
    add_subdirectory("${change_root}" sixdb)
endif()

target_link_libraries(frame_change_envelope_check PRIVATE ikea::seriespack ikea::tuplepack)

target_link_options(frame_change_envelope_sanitize PRIVATE -fsanitize=address,undefined)

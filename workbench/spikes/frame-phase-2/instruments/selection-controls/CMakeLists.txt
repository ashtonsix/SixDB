if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  cmake_minimum_required(VERSION 3.24)
  find_program(selection_clang NAMES clang-21 REQUIRED)
  find_program(selection_clangxx NAMES clang++-21 REQUIRED)
  set(CMAKE_C_COMPILER "${selection_clang}" CACHE FILEPATH "Pinned compiler")
  set(CMAKE_CXX_COMPILER "${selection_clangxx}" CACHE FILEPATH "Pinned compiler")
  project(FrameSelectionControls LANGUAGES C CXX)
  if(NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "21.1.8")
    message(FATAL_ERROR "Controls require pinned Clang 21.1.8")
  endif()
  set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
  set(SIXDB_MARCH "armv8-a+simd" CACHE STRING "ISA")
  set(sixdb_clang_tune "neoverse-v2")
endif()
get_filename_component(selection_root "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
include("${selection_root}/workbench/peers/frame-selection.cmake")
foreach(peer IN ITEMS roaring vqsort)
  add_executable(frame_${peer}_check ${peer}_check.cpp)
  target_compile_features(frame_${peer}_check PRIVATE cxx_std_23)
  workbench_selection_settings(frame_${peer}_check)
  if(SIXDB_FRAME_SELECTION_SANITIZE)
    if(NOT SIXDB_FRAME_SELECTION_SANITIZERS)
      set(SIXDB_FRAME_SELECTION_SANITIZERS "address,undefined")
    endif()
    target_link_options(frame_${peer}_check PRIVATE "-fsanitize=${SIXDB_FRAME_SELECTION_SANITIZERS}")
  endif()
endforeach()
target_link_libraries(frame_roaring_check PRIVATE workbench::croaring)
target_link_libraries(frame_vqsort_check PRIVATE workbench::vqsort)

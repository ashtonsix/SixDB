if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  cmake_minimum_required(VERSION 3.24)
  find_program(calico_clangxx NAMES clang++-21 REQUIRED)
  set(CMAKE_CXX_COMPILER "${calico_clangxx}" CACHE FILEPATH "Pinned compiler")
  project(FrameCalicoControl LANGUAGES CXX)
  set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
  set(SIXDB_MARCH "armv8-a+simd" CACHE STRING "ISA")
  set(sixdb_clang_tune "neoverse-v2")
endif()
get_filename_component(calico_root "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
include("${calico_root}/workbench/peers/frame-calico.cmake")
add_library(frame_calico_controls STATIC EXCLUDE_FROM_ALL control.cpp corpus_sum.cpp)
target_include_directories(frame_calico_controls PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(frame_calico_controls PUBLIC workbench::calico_frame)
add_executable(frame_calico_check EXCLUDE_FROM_ALL check.cpp)
target_link_libraries(frame_calico_check PRIVATE frame_calico_controls)
add_executable(frame_calico_corpus_check EXCLUDE_FROM_ALL corpus_check.cpp)
target_link_libraries(frame_calico_corpus_check PRIVATE frame_calico_controls)
foreach(target IN ITEMS frame_calico_controls frame_calico_check frame_calico_corpus_check)
  target_compile_features(${target} PUBLIC cxx_std_23)
  set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
  target_compile_options(${target} PRIVATE -O3 -gline-tables-only -fno-fast-math -ffp-contract=off -fdenormal-fp-math=ieee)
  if(SIXDB_MARCH)
    target_compile_options(${target} PRIVATE "-march=${SIXDB_MARCH}")
  endif()
  if(sixdb_clang_tune)
    target_compile_options(${target} PRIVATE "-mtune=${sixdb_clang_tune}")
  endif()
  if(SIXDB_FRAME_CALICO_SANITIZE)
    target_compile_options(${target} PRIVATE -O1 -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  endif()
endforeach()

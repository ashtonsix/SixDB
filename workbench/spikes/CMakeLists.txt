set(SIXDB_SPIKES "" CACHE STRING "Semicolon-separated spike directory names to configure")
foreach(spike IN LISTS SIXDB_SPIKES)
  if(NOT spike MATCHES "^[A-Za-z0-9][A-Za-z0-9_-]*$")
    message(FATAL_ERROR "Invalid spike name: ${spike}; use a direct child directory name")
  endif()
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${spike}/CMakeLists.txt")
    message(FATAL_ERROR "Spike ${spike} needs workbench/spikes/${spike}/CMakeLists.txt")
  endif()
  add_subdirectory("${spike}" EXCLUDE_FROM_ALL)
endforeach()

cmake_minimum_required(VERSION 3.24)
project(heap_control_checks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
set(RUNTIME_SOURCE "" CACHE PATH "Frozen patched Orbital source directory")
if(NOT EXISTS "${RUNTIME_SOURCE}/src/allocation/finite.cpp")
  message(FATAL_ERROR "Set RUNTIME_SOURCE to the patched frozen orbital directory")
endif()
add_compile_options(-O3 -g -fno-fast-math -ffp-contract=off -Wall -Wextra -Wpedantic)
add_library(orbital STATIC)
foreach(tu types.cpp worker.cpp allocation/allocation.cpp allocation/arena.cpp
    allocation/finite.cpp allocation/group.cpp allocation/window.cpp)
  target_sources(orbital PRIVATE "${RUNTIME_SOURCE}/src/${tu}")
endforeach()
target_include_directories(orbital PUBLIC "${RUNTIME_SOURCE}/include"
    "${RUNTIME_SOURCE}/src/allocation")
target_link_libraries(orbital PUBLIC Threads::Threads)
foreach(probe smoke census)
  add_executable(heap_only_${probe} probes/${probe}.cpp)
  target_include_directories(heap_only_${probe} PRIVATE ../../shape)
  target_link_libraries(heap_only_${probe} PRIVATE orbital)
endforeach()

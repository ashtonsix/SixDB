cmake_minimum_required(VERSION 3.24)
if(NOT COHESION_SOURCE_ROOT OR NOT DEFINED COHESION_BEFORE)
  message(FATAL_ERROR "Supply frozen COHESION_SOURCE_ROOT and COHESION_BEFORE=0/1")
endif()
include("${COHESION_SOURCE_ROOT}/cmake/ClangVersion.cmake")
if(NOT DEFINED CMAKE_CXX_COMPILER)
  find_program(COHESION_CLANGXX NAMES clang++-${SIXDB_CLANG_MAJOR} clang++ REQUIRED)
  set(CMAKE_CXX_COMPILER "${COHESION_CLANGXX}" CACHE FILEPATH "Pinned compiler")
endif()
# Match SixDB before CMake initializes its default Release=NDEBUG flags.
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -g" CACHE STRING "Release flags")
project(IkeaCohesion LANGUAGES CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_subdirectory("${COHESION_SOURCE_ROOT}" sixdb)
add_executable(ikea_cohesion_initialize ../cohesion-initialize.cpp)
sixdb_target(ikea_cohesion_initialize)
target_compile_definitions(ikea_cohesion_initialize PRIVATE COHESION_BEFORE=${COHESION_BEFORE})
target_link_libraries(ikea_cohesion_initialize PRIVATE ikea::tuplepack benchmark::benchmark)

add_executable(ikea_predicate_blocks ../handle-boundaries/blocks.cpp)
sixdb_target(ikea_predicate_blocks)
target_link_libraries(ikea_predicate_blocks PRIVATE ikea::tuplepack benchmark::benchmark)

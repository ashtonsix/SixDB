cmake_minimum_required(VERSION 3.24)
get_filename_component(SIXDB_CONTROL_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
include("${SIXDB_CONTROL_SOURCE_ROOT}/cmake/ClangVersion.cmake")
if(NOT DEFINED CMAKE_CXX_COMPILER)
  find_program(CONTROL_CLANGXX NAMES clang++-${SIXDB_CLANG_MAJOR} clang++ REQUIRED)
  set(CMAKE_CXX_COMPILER "${CONTROL_CLANGXX}" CACHE FILEPATH "Pinned compiler")
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -g" CACHE STRING "Release flags")
project(IkeaExecutionControls LANGUAGES CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_subdirectory("${SIXDB_CONTROL_SOURCE_ROOT}" sixdb)
add_executable(ikea_predicate_direct_commands commands.cpp)
sixdb_target(ikea_predicate_direct_commands)
target_link_libraries(ikea_predicate_direct_commands PRIVATE ikea::tuplepack benchmark::benchmark)
option(IKEA_CONTROL_SERIES "Build the shared SeriesPack control comparison" OFF)
if(IKEA_CONTROL_SERIES)
  add_subdirectory(../series-controls series-controls)
endif()

find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(SWISS_BATCH_BUILD_PEERS "Build pinned native Boost/Abseil controls" ON)
if(SWISS_BATCH_BUILD_PEERS)
  include("${PROJECT_SOURCE_DIR}/workbench/benchmarks/hashtable/abseil.cmake")
  include(FetchContent)
  FetchContent_Declare(swiss_boost
    URL https://archives.boost.io/release/1.92.0/source/boost_1_92_0.tar.bz2
    URL_HASH SHA256=5c1d40cb8e19adbf740a4ec2da35b3e58f3f5804b1dce44deb53df72193cbc6c
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE SOURCE_SUBDIR no-cmake-needed)
  FetchContent_MakeAvailable(swiss_boost)
endif()
set(SWISS_BATCH_VARIANTS "group16;batch64;tail64" CACHE STRING "Private Swiss algorithms")
set(variants ${SWISS_BATCH_VARIANTS})
option(SWISS_BATCH_READ_ONLY "Keep the current scalar write engine" ON)
set(read_only_option)
if(SWISS_BATCH_READ_ONLY)
  set(read_only_option --read-only)
endif()
foreach(variant IN LISTS variants)
  set(overlay "${CMAKE_CURRENT_BINARY_DIR}/${variant}/include")
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/overlay.py"
    --source "${PROJECT_SOURCE_DIR}" --output "${overlay}" --variant "${variant}" ${read_only_option}
    COMMAND_ERROR_IS_FATAL ANY)
  add_library(swiss_${variant} INTERFACE)
  target_include_directories(swiss_${variant} INTERFACE "${overlay}")
  target_link_libraries(swiss_${variant} INTERFACE ikea::hashtable)
  foreach(kind check codegen)
    add_executable(swiss_${kind}_${variant} ${kind}.cpp)
    sixdb_target(swiss_${kind}_${variant})
    target_link_libraries(swiss_${kind}_${variant} PRIVATE swiss_${variant})
    target_include_directories(swiss_${kind}_${variant} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_options(swiss_${kind}_${variant} PRIVATE -mllvm -inline-threshold=1000 -falign-functions=64)
  endforeach()
endforeach()
foreach(variant IN LISTS variants)
  add_executable(swiss_trace_${variant} trace.cpp)
  sixdb_target(swiss_trace_${variant})
  target_link_libraries(swiss_trace_${variant} PRIVATE swiss_${variant})
  target_compile_definitions(swiss_trace_${variant} PRIVATE SIXDB_SWISS_TRACE=1)
endforeach()
add_executable(swiss_mask_check mask-check.cpp)
sixdb_target(swiss_mask_check)
target_link_libraries(swiss_mask_check PRIVATE ikea::hashtable)
if(SWISS_BATCH_BUILD_PEERS)
  list(APPEND variants peers)
endif()
set(SWISS_BATCH_CASES "30_10_0;30_10_8;30_10_64;30_20_0;30_20_8" CACHE STRING "K_b_V cases")
foreach(case IN LISTS SWISS_BATCH_CASES)
  string(REPLACE "_" ";" fields "${case}")
  list(GET fields 0 k)
  list(GET fields 1 b)
  list(GET fields 2 v)
  foreach(variant IN LISTS variants)
    set(width 16)
    set(target swiss_${variant}_${case})
    add_executable(${target} bench.cpp)
    sixdb_target(${target})
    target_compile_definitions(${target} PRIVATE NDEBUG OVERFLOW_K=${k} OVERFLOW_B=${b}
      OVERFLOW_V=${v} OVERFLOW_NAME="${variant}" OVERFLOW_PEERS=$<STREQUAL:${variant},peers>
      OVERFLOW_SWISS_CEILING32=28 SWISS_GROUP_BYTES=${width})
    target_compile_options(${target} PRIVATE -mllvm -inline-threshold=1000 -falign-functions=64)
    if(variant STREQUAL peers)
      target_link_libraries(${target} PRIVATE ikea::hashtable hashtable_abseil_baseline)
      target_include_directories(${target} SYSTEM PRIVATE "${swiss_boost_SOURCE_DIR}")
    else()
      target_link_libraries(${target} PRIVATE swiss_${variant})
    endif()
  endforeach()
endforeach()

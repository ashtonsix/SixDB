  execute_process(COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/original_overlay.py"
    "${V3_BEC_BASELINE_ARCHIVE}" "${CMAKE_CURRENT_BINARY_DIR}/original-bec" COMMAND_ERROR_IS_FATAL ANY)
  target_include_directories(ikea_bec256 BEFORE PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/original-bec")

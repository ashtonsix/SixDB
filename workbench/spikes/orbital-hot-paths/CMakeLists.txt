# Each variants/NAME.patch applies to a configure-time copy of Orbital's sources
# and headers. The variant binary runs the allocation, Lane and catalog cases
# against that copy; production sources are never modified.
find_package(Git REQUIRED)
find_package(Threads REQUIRED)
file(GLOB hot_variants CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/variants/*.patch")
set(hot_applied)
foreach(patch IN LISTS hot_variants)
  get_filename_component(variant "${patch}" NAME_WE)
  set(root "${CMAKE_CURRENT_BINARY_DIR}/${variant}")
  file(REMOVE_RECURSE "${root}")
  file(COPY "${PROJECT_SOURCE_DIR}/orbital/src" DESTINATION "${root}/orbital")
  file(COPY "${PROJECT_SOURCE_DIR}/orbital/include" DESTINATION "${root}/orbital")
  execute_process(COMMAND "${GIT_EXECUTABLE}" apply --unsafe-paths "--directory=${root}" "${patch}"
    RESULT_VARIABLE applied)
  if(NOT applied EQUAL 0)
    # A variant made against an older Orbital is skipped, not silently rebased.
    message(WARNING "orbital-hot-paths: ${variant}.patch does not apply to this Orbital; skipped")
    continue()
  endif()
  list(APPEND hot_applied "${patch}")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${patch}")
  file(GLOB_RECURSE sources "${root}/orbital/src/*.cpp")
  add_library(orbital_hot_${variant} STATIC ${sources})
  sixdb_target(orbital_hot_${variant})
  target_include_directories(orbital_hot_${variant} PUBLIC "${root}/orbital/include")
  target_link_libraries(orbital_hot_${variant} PUBLIC Threads::Threads)
  add_executable(orbital_hot_paths_${variant} bench.cpp "${PROJECT_SOURCE_DIR}/orbital/test/object_material.cpp")
  sixdb_target(orbital_hot_paths_${variant})
  target_compile_definitions(orbital_hot_paths_${variant} PRIVATE SIXDB_HOT_FULL=0 SIXDB_HOT_OBJECTS=1)
  target_include_directories(orbital_hot_paths_${variant} PRIVATE "${PROJECT_SOURCE_DIR}/orbital/test")
  target_link_libraries(orbital_hot_paths_${variant} PRIVATE orbital_hot_${variant})
  add_executable(orbital_hot_census_${variant} census.cpp)
  sixdb_target(orbital_hot_census_${variant})
  target_link_libraries(orbital_hot_census_${variant} PRIVATE orbital_hot_${variant})
endforeach()

# Rebuild Orbital's own allocation/worker checks against each variant: same
# test sources, include roles and link wrappers, with the variant library.
set(hot_checks allocation finite class reclamation window window_index live_tail
  return worker lifetime thread preparation notification budget
  object history_reclamation directory extension representation part_sources)
foreach(patch IN LISTS hot_applied)
  get_filename_component(variant "${patch}" NAME_WE)
  set(root "${CMAKE_CURRENT_BINARY_DIR}/${variant}")
  foreach(check IN LISTS hot_checks)
    set(original orbital_${check}_check)
    get_target_property(sources ${original} SOURCES)
    get_target_property(includes ${original} INCLUDE_DIRECTORIES)
    get_target_property(options ${original} LINK_OPTIONS)
    get_target_property(libraries ${original} LINK_LIBRARIES)
    list(TRANSFORM sources PREPEND "${PROJECT_SOURCE_DIR}/orbital/")
    set(target orbital_hot_${variant}_${check}_check)
    add_executable(${target} EXCLUDE_FROM_ALL ${sources})
    sixdb_target(${target})
    foreach(directory IN LISTS includes)
      if(directory MATCHES "orbital/src")
        string(REPLACE "${PROJECT_SOURCE_DIR}/orbital" "${root}/orbital" directory "${directory}")
        target_include_directories(${target} BEFORE PRIVATE "${directory}")
      endif()
    endforeach()
    target_include_directories(${target} PRIVATE "${PROJECT_SOURCE_DIR}/orbital/test")
    list(FILTER libraries EXCLUDE REGEX "^(orbital|orbital::orbital|sixdb::build_options)$")
    target_link_libraries(${target} PRIVATE orbital_hot_${variant} ${libraries})
    if(options)
      target_link_options(${target} PRIVATE ${options})
    endif()
    add_test(NAME ${target} COMMAND ${target})
    set_tests_properties(${target} PROPERTIES LABELS "orbital-hot-${variant}" TIMEOUT 120)
  endforeach()
  add_custom_target(orbital_hot_${variant}_validate
    COMMAND "${CMAKE_CTEST_COMMAND}" -L "orbital-hot-${variant}" --output-on-failure
    WORKING_DIRECTORY "${PROJECT_BINARY_DIR}")
  foreach(check IN LISTS hot_checks)
    add_dependencies(orbital_hot_${variant}_validate orbital_hot_${variant}_${check}_check)
  endforeach()
endforeach()

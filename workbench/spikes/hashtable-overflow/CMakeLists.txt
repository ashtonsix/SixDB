find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(OVERFLOW_BUILD_PEERS "Acquire/build the native product controls" ON)
option(OVERFLOW_DENSITY "Build the 93.75/96.875 percent Swiss variants" OFF)
if(OVERFLOW_BUILD_PEERS)
include("${PROJECT_SOURCE_DIR}/workbench/benchmarks/hashtable/abseil.cmake")
include(FetchContent)
FetchContent_Declare(overflow_boost
  URL https://archives.boost.io/release/1.92.0/source/boost_1_92_0.tar.bz2
  URL_HASH SHA256=5c1d40cb8e19adbf740a4ec2da35b3e58f3f5804b1dce44deb53df72193cbc6c
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE SOURCE_SUBDIR no-cmake-needed)
FetchContent_MakeAvailable(overflow_boost)
endif()

foreach(variant baseline tag8 filter15)
  set(overlay "${CMAKE_CURRENT_BINARY_DIR}/${variant}/include")
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/overlay.py"
    --source "${PROJECT_SOURCE_DIR}" --output "${overlay}" --variant "${variant}"
    COMMAND_ERROR_IS_FATAL ANY)
  add_library(overflow_${variant} INTERFACE)
  target_include_directories(overflow_${variant} INTERFACE "${overlay}")
  target_link_libraries(overflow_${variant} INTERFACE ikea::hashtable)
  add_executable(overflow_check_${variant} check.cpp)
  sixdb_target(overflow_check_${variant})
  target_link_libraries(overflow_check_${variant} PRIVATE overflow_${variant})
  add_executable(overflow_codegen_${variant} codegen.cpp)
  sixdb_target(overflow_codegen_${variant})
  target_link_libraries(overflow_codegen_${variant} PRIVATE overflow_${variant})
  target_compile_options(overflow_codegen_${variant} PRIVATE -mllvm -inline-threshold=1000 -falign-functions=64)
endforeach()
set(OVERFLOW_CASES "30_10_0;30_10_8;30_10_64;14_10_0;14_10_8;30_20_0;30_20_8"
    CACHE STRING "K_b_V geometries")
set(variants baseline tag8 filter15)
if(OVERFLOW_BUILD_PEERS)
  list(APPEND variants peers)
endif()
if(OVERFLOW_DENSITY)
  list(APPEND variants tag8_c30 tag8_c31 filter15_c30 filter15_c31)
endif()
foreach(case IN LISTS OVERFLOW_CASES)
  string(REPLACE "_" ";" fields "${case}")
  list(GET fields 0 k)
  list(GET fields 1 b)
  list(GET fields 2 v)
  foreach(variant IN LISTS variants)
    set(base "${variant}")
    set(ceiling 28)
    if(variant MATCHES "^(tag8|filter15)_c(30|31)$")
      set(base "${CMAKE_MATCH_1}")
      set(ceiling "${CMAKE_MATCH_2}")
    endif()
    set(target overflow_${variant}_${case})
    add_executable(${target} bench.cpp)
    sixdb_target(${target})
    target_compile_definitions(${target} PRIVATE NDEBUG OVERFLOW_K=${k} OVERFLOW_B=${b}
      OVERFLOW_V=${v} OVERFLOW_NAME="${variant}" OVERFLOW_PEERS=$<STREQUAL:${variant},peers>
      OVERFLOW_SWISS_CEILING32=${ceiling})
    target_compile_options(${target} PRIVATE -mllvm -inline-threshold=1000 -falign-functions=64)
    if(variant STREQUAL peers)
      target_link_libraries(${target} PRIVATE ikea::hashtable hashtable_abseil_baseline)
      target_include_directories(${target} SYSTEM PRIVATE "${overflow_boost_SOURCE_DIR}")
    else()
      target_link_libraries(${target} PRIVATE overflow_${base})
    endif()
  endforeach()
endforeach()

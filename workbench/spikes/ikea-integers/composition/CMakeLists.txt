if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  cmake_minimum_required(VERSION 3.24)
  project(ikea_integer_composition LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 23)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
  set(CMAKE_UNITY_BUILD OFF)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "21.1.8")
    message(FATAL_ERROR "Use pinned Linux Clang 21.1.8")
  endif()
  function(sixdb_target target)
    target_compile_options(${target} PRIVATE -UNDEBUG -fno-fast-math -ffp-contract=off)
  endfunction()
endif()

add_library(ikea_integer_composition STATIC prepare.cpp read_regions.cpp filter.cpp sum.cpp driver.cpp)
sixdb_target(ikea_integer_composition)
target_include_directories(ikea_integer_composition PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
  target_compile_options(ikea_integer_composition PUBLIC -mavx2)
endif()
add_executable(ikea_integer_composition_check check.cpp)
sixdb_target(ikea_integer_composition_check)
target_link_libraries(ikea_integer_composition_check PRIVATE ikea_integer_composition)
add_executable(ikea_integer_composition_bench bench.cpp)
sixdb_target(ikea_integer_composition_bench)
target_link_libraries(ikea_integer_composition_bench PRIVATE ikea_integer_composition)

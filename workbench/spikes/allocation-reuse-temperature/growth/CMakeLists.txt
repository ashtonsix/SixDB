cmake_minimum_required(VERSION 3.24)
set(SIXDB_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../..")
include("${SIXDB_ROOT}/cmake/ClangVersion.cmake")
if(NOT DEFINED CMAKE_CXX_COMPILER)
  find_program(CMAKE_CXX_COMPILER NAMES clang++-21 REQUIRED)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -g" CACHE STRING "Release flags")
project(WindowGrowth LANGUAGES CXX)
if(NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL SIXDB_CLANG_VERSION)
  message(FATAL_ERROR "Requires pinned Clang ${SIXDB_CLANG_VERSION}")
endif()
include("${SIXDB_ROOT}/cmake/Targets.cmake")
find_package(Threads REQUIRED)
set(orbital_sources types.cpp worker.cpp platform.cpp object/storage.cpp object/representation.cpp
  allocation/allocation.cpp allocation/arena.cpp allocation/group.cpp allocation/finite.cpp)
list(TRANSFORM orbital_sources PREPEND "${SIXDB_ROOT}/orbital/src/")
# Each captured version configures a separate build. Share common TUs only
# between that version's timing and counter executables, never across layouts.
add_library(common OBJECT ${orbital_sources})
sixdb_target(common)
target_include_directories(common PUBLIC "${SIXDB_ROOT}/orbital/include")
foreach(mode timing counters)
  set(window "${SIXDB_ROOT}/orbital/src/allocation/window.cpp")
  if(mode STREQUAL "counters")
    set(window "${GROWTH_COUNTER_WINDOW}")
  endif()
  add_library(storage_${mode} STATIC "${window}" $<TARGET_OBJECTS:common>)
  sixdb_target(storage_${mode})
  target_include_directories(storage_${mode} PUBLIC "${SIXDB_ROOT}/orbital/include"
    PRIVATE "${SIXDB_ROOT}/orbital/src/allocation")
  target_link_libraries(storage_${mode} PUBLIC Threads::Threads)
  add_executable(growth_${mode} harness.cpp)
  sixdb_target(growth_${mode})
  target_include_directories(growth_${mode} PRIVATE "${SIXDB_ROOT}/orbital/src/allocation")
  target_compile_definitions(growth_${mode} PRIVATE GROWTH_HAS_INDEX=${GROWTH_HAS_INDEX})
  if(mode STREQUAL "counters")
    target_compile_definitions(growth_${mode} PRIVATE GROWTH_COUNTERS=1)
  endif()
  target_link_libraries(growth_${mode} PRIVATE storage_${mode})
endforeach()

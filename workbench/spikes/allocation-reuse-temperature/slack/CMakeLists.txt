cmake_minimum_required(VERSION 3.24)
set(SIXDB_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../..")
include("${SIXDB_ROOT}/cmake/ClangVersion.cmake")
if(NOT DEFINED CMAKE_CXX_COMPILER)
  find_program(CMAKE_CXX_COMPILER NAMES clang++-21 REQUIRED)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -g" CACHE STRING "Release flags")
project(WindowReuse LANGUAGES CXX)
if(NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL SIXDB_CLANG_VERSION)
  message(FATAL_ERROR "Requires pinned Clang ${SIXDB_CLANG_VERSION}")
endif()
include("${SIXDB_ROOT}/cmake/Targets.cmake")
find_package(Threads REQUIRED)
set(orbital_sources
  types.cpp worker.cpp platform.cpp object/storage.cpp object/representation.cpp
  allocation/allocation.cpp allocation/arena.cpp allocation/group.cpp allocation/finite.cpp)
list(TRANSFORM orbital_sources PREPEND "${SIXDB_ROOT}/orbital/src/")
add_library(common OBJECT ${orbital_sources})
sixdb_target(common)
target_include_directories(common PUBLIC "${SIXDB_ROOT}/orbital/include")
foreach(policy baseline prefer-2x best-fit)
  add_library(storage_${policy} STATIC "${WINDOW_VARIANTS}/${policy}.cpp" $<TARGET_OBJECTS:common>)
  sixdb_target(storage_${policy})
  target_include_directories(storage_${policy} PUBLIC "${SIXDB_ROOT}/orbital/include"
    PRIVATE "${SIXDB_ROOT}/orbital/src/allocation")
  target_link_libraries(storage_${policy} PUBLIC Threads::Threads)
  add_executable(reuse_${policy} harness.cpp)
  sixdb_target(reuse_${policy})
  target_include_directories(reuse_${policy} PRIVATE "${SIXDB_ROOT}/orbital/src/allocation")
  target_link_libraries(reuse_${policy} PRIVATE storage_${policy})
  target_link_options(reuse_${policy} PRIVATE -Wl,--wrap=mmap,--wrap=mprotect,--wrap=munmap)
  add_executable(check_${policy} "${SIXDB_ROOT}/orbital/test/window.cpp")
  sixdb_target(check_${policy})
  target_include_directories(check_${policy} PRIVATE "${SIXDB_ROOT}/orbital/src/allocation")
  target_link_libraries(check_${policy} PRIVATE storage_${policy})
  target_link_options(check_${policy} PRIVATE -Wl,--wrap=mprotect)
endforeach()
add_executable(tail_probe tail_probe.cpp)
sixdb_target(tail_probe)
target_link_libraries(tail_probe PRIVATE storage_baseline)

function(loom_consumer_settings target kind scale)
  sixdb_target(${target})
  target_link_libraries(${target} PRIVATE hashtable_abseil_baseline
    ikea::seriespack ikea::tuplepack ikea::streampack orbital::orbital loom::loom)
  target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}
    ${consumer_root}/workbench/spikes/hashtable-design)
  target_compile_definitions(${target} PRIVATE NDEBUG HT_STAGED_KIND=${kind}
    HT_STAGED_WIDTH=8 HT_STAGED_KEY=40 HT_STAGED_GRAPH_HINT=1)
  target_compile_options(${target} PRIVATE -fno-vectorize -fno-slp-vectorize)
  if(scale STREQUAL "large")
    if(kind EQUAL 0)
      target_compile_definitions(${target} PRIVATE HT_STAGED_A=20 HT_STAGED_UNIT=64)
    else()
      target_compile_definitions(${target} PRIVATE HT_STAGED_A=19)
    endif()
  elseif(kind EQUAL 0)
    target_compile_definitions(${target} PRIVATE HT_STAGED_A=9 HT_STAGED_UNIT=64)
  endif()
endfunction()

cmake_minimum_required(VERSION 3.24)
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -g" CACHE STRING "Release flags")
project(IndexDirectoryCalico LANGUAGES C CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Calico requires Linux")
endif()
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR
    NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "21.1.8")
  message(FATAL_ERROR "Use SixDB's pinned Clang 21.1.8")
endif()
set(CALICO_SOURCE "" CACHE PATH "Existing Calico source checkout or captured source")
if(NOT EXISTS "${CALICO_SOURCE}/calico/include/calico/runtime.h")
  message(FATAL_ERROR "Set CALICO_SOURCE to the actual Calico source directory")
endif()
# Calico owns its native dependency graph, release policy and public API.
# EXCLUDE_FROM_ALL avoids building its tests, rigs and unrelated executables.
add_subdirectory("${CALICO_SOURCE}" calico EXCLUDE_FROM_ALL)
add_executable(index_directory_calico main.cpp)
target_compile_features(index_directory_calico PRIVATE cxx_std_23)
set_target_properties(index_directory_calico PROPERTIES CXX_EXTENSIONS OFF)
target_compile_options(index_directory_calico PRIVATE -Wall -Wextra -Wpedantic
    -Wno-unused-parameter -DNDEBUG -DNONOPT_FSST)
if(CALICO_NATIVE)
  target_compile_options(index_directory_calico PRIVATE -march=native)
endif()
target_link_libraries(index_directory_calico PRIVATE calico::runtime)

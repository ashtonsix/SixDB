if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The first directory assembly prototype uses the Linux ELF ABI")
endif()
set(CMAKE_ASM_COMPILER "${CMAKE_CXX_COMPILER}")
set(CMAKE_ASM_COMPILER_TARGET "${CMAKE_CXX_COMPILER_TARGET}")
enable_language(ASM)
option(INDEX_DIRECTORY_SCAN "Use current Ikea full-SIMD scan rather than direct boundary" OFF)
set(INDEX_DIRECTORY_LEAF_BOUND 32 CACHE STRING "Admitted maximum leaf rank count;0 is unconstrained control")
set(INDEX_DIRECTORY_INTERNAL_BOUND 32 CACHE STRING "Admitted maximum internal rank count;0 is unconstrained control")
add_library(index_directory_asm STATIC asm/exact.S asm/group16.S asm/payload.S asm/replace.S
    src/ikea_calls.cpp src/read.cpp src/material.cpp src/group_fixture.cpp)
sixdb_target(index_directory_asm)
target_include_directories(index_directory_asm PUBLIC include)
target_link_libraries(index_directory_asm PUBLIC orbital::orbital loom::loom ikea::boundlist)
target_compile_definitions(index_directory_asm PRIVATE INDEX_DIRECTORY_SCAN=$<BOOL:${INDEX_DIRECTORY_SCAN}>)
target_compile_definitions(index_directory_asm PRIVATE
    INDEX_DIRECTORY_LEAF_BOUND=${INDEX_DIRECTORY_LEAF_BOUND}
    INDEX_DIRECTORY_INTERNAL_BOUND=${INDEX_DIRECTORY_INTERNAL_BOUND})
set_source_files_properties(src/ikea_calls.cpp PROPERTIES
    COMPILE_OPTIONS "-fno-vectorize;-fno-slp-vectorize")
add_executable(index_directory_check checks/selected.cpp)
sixdb_target(index_directory_check)
target_link_libraries(index_directory_check PRIVATE index_directory_asm)
add_test(NAME index_directory_check COMMAND index_directory_check)
add_executable(index_directory_material_check checks/material.cpp)
sixdb_target(index_directory_material_check)
target_link_libraries(index_directory_material_check PRIVATE index_directory_asm)
add_test(NAME index_directory_material_check COMMAND index_directory_material_check)
add_executable(index_directory_composed_check checks/composed.cpp)
sixdb_target(index_directory_composed_check)
target_link_libraries(index_directory_composed_check PRIVATE index_directory_asm)
add_test(NAME index_directory_composed_check COMMAND index_directory_composed_check)
include("${PROJECT_SOURCE_DIR}/workbench/peers/lmdb.cmake")
add_executable(index_directory_driver src/driver.cpp peers/lmdb.cpp)
sixdb_target(index_directory_driver)
target_link_libraries(index_directory_driver PRIVATE index_directory_asm workbench::lmdb)

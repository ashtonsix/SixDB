set(SIXDB_PROTOTYPES "" CACHE STRING "Semicolon-separated prototype directory names to configure")
foreach(prototype IN LISTS SIXDB_PROTOTYPES)
  if(NOT prototype MATCHES "^[A-Za-z0-9][A-Za-z0-9_-]*$")
    message(FATAL_ERROR "Invalid prototype name: ${prototype}; use a direct child directory name")
  endif()
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${prototype}/CMakeLists.txt")
    message(FATAL_ERROR "Prototype ${prototype} needs workbench/prototypes/${prototype}/CMakeLists.txt")
  endif()
  add_subdirectory("${prototype}" EXCLUDE_FROM_ALL)
endforeach()

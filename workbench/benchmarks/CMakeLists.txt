set(SIXDB_BENCHMARKS "" CACHE STRING "Semicolon-separated recurring benchmark suites to configure")
foreach(suite IN LISTS SIXDB_BENCHMARKS)
  if(NOT suite MATCHES "^[A-Za-z0-9][A-Za-z0-9_-]*$")
    message(FATAL_ERROR "Invalid benchmark suite name: ${suite}; use a direct child directory name")
  endif()
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${suite}/CMakeLists.txt")
    message(FATAL_ERROR "Benchmark ${suite} needs workbench/benchmarks/${suite}/CMakeLists.txt")
  endif()
  add_subdirectory("${suite}" EXCLUDE_FROM_ALL)
endforeach()

include("${CMAKE_CURRENT_LIST_DIR}/abseil.cmake")
set(IKEA_HASHTABLE_W 24 CACHE STRING "Read comparison key width")
set(IKEA_HASHTABLE_B 10 CACHE STRING "Read comparison bucket bits")
add_library(ikea_hashtable_benchmark_support STATIC EXCLUDE_FROM_ALL common.cpp)
sixdb_target(ikea_hashtable_benchmark_support)
target_link_libraries(ikea_hashtable_benchmark_support PUBLIC ikea::hashtable)
add_executable(ikea_hashtable_reads EXCLUDE_FROM_ALL reads.cpp)
sixdb_target(ikea_hashtable_reads)
target_compile_definitions(ikea_hashtable_reads PRIVATE NDEBUG
  IKEA_HASHTABLE_W=${IKEA_HASHTABLE_W} IKEA_HASHTABLE_B=${IKEA_HASHTABLE_B})
target_compile_options(ikea_hashtable_reads PRIVATE -mllvm -inline-threshold=1000 -falign-functions=64)
target_link_libraries(ikea_hashtable_reads PRIVATE ikea_hashtable_benchmark_support hashtable_abseil_baseline)
foreach(k IN ITEMS 14 30)
  foreach(v IN ITEMS 0 8 64)
    foreach(b IN ITEMS 10 18)
      math(EXPR w "${k}+${b}")
      set(target ikea_hashtable_operations_k${k}_b${b}_v${v})
      add_executable(${target} EXCLUDE_FROM_ALL operations.cpp)
      sixdb_target(${target})
      target_compile_definitions(${target} PRIVATE NDEBUG
        IKEA_HASHTABLE_W=${w} IKEA_HASHTABLE_B=${b} IKEA_HASHTABLE_V=${v})
      target_compile_options(${target} PRIVATE -mllvm -inline-threshold=1000 -falign-functions=64)
      target_link_libraries(${target} PRIVATE ikea_hashtable_benchmark_support hashtable_abseil_baseline)
    endforeach()
  endforeach()
endforeach()

set(IKEA_ACCUMULATION_REDUCER_KIND 0 CACHE STRING "0 byte fold,1 copy,2 word add,3 word fold")
foreach(panel basic wide)
  foreach(v IN ITEMS 0 8 64)
    set(target ikea_hashtable_accumulation_${panel}_${v})
    add_executable(${target} EXCLUDE_FROM_ALL accumulation.cpp)
    sixdb_target(${target})
    target_compile_definitions(${target} PRIVATE NDEBUG MERGE_VALUE_BYTES=${v}
      MERGE_REDUCER_KIND=${IKEA_ACCUMULATION_REDUCER_KIND}
      MERGE_WIDE_FAMILIES=$<STREQUAL:${panel},wide>)
    target_compile_options(${target} PRIVATE -mllvm -inline-threshold=1000 -falign-functions=64)
    target_link_libraries(${target} PRIVATE ikea_hashtable_benchmark_support hashtable_abseil_baseline)
  endforeach()
endforeach()
